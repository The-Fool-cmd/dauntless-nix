# CMake build of MysticParadox.dll, equivalent to the Release|x64 configuration of MysticParadox.vcxproj.
# Added by dauntless-nix so the runtime can be cross-built without Visual Studio (clang-cl + lld-link + the MSVC
# CRT/Windows SDK; see nix/pkgs/msvc-toolchain.nix). Native MSVC + CMake works too. The .vcxproj stays the reference.
#
#   cmake -G Ninja -DCMAKE_TOOLCHAIN_FILE=<toolchain> -DMP_SDK_DIR=<CppSDK dir> -DMP_PUBLIC_HOST=<host> <this dir>
cmake_minimum_required(VERSION 3.20)
project(MysticParadox LANGUAGES C CXX RC)

set(MP_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH
    "Directory holding the Dumper-7 CppSDK output (SDK.hpp and SDK/). Default: this folder, as in the .vcxproj.")
set(MP_PUBLIC_HOST "" CACHE STRING
    "Hostname compiled in as MP_PUBLIC_HOST. Empty: use deployment_config.generated.h from this folder.")

if(NOT EXISTS "${MP_SDK_DIR}/SDK.hpp" OR NOT IS_DIRECTORY "${MP_SDK_DIR}/SDK")
  message(FATAL_ERROR "MP_SDK_DIR=${MP_SDK_DIR} has no SDK.hpp/SDK. Generate it with Dumper-7 (docs/GENERATING_SDK.md).")
endif()

# deployment_config.generated.h: generated from MP_PUBLIC_HOST into the build tree, or the hand-made one beside the
# sources. The build-tree include directory comes first so a generated header wins.
set(MP_GENERATED_INCLUDE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(MP_PUBLIC_HOST)
  file(CONFIGURE OUTPUT "${MP_GENERATED_INCLUDE}/deployment_config.generated.h" CONTENT
"#pragma once
// Generated by ParadoxRuntime/CMakeLists.txt from MP_PUBLIC_HOST. Do not commit.
#define MP_PUBLIC_HOST L\"@MP_PUBLIC_HOST@\"
" @ONLY)
elseif(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/deployment_config.generated.h")
  message(FATAL_ERROR "Set -DMP_PUBLIC_HOST=<host> or create deployment_config.generated.h (see the .example).")
endif()

# The same SDK translation units the .vcxproj compiles (not every *_functions.cpp of the dump).
set(MP_SDK_SOURCES
  Archon_functions.cpp Basic.cpp BPGM_Archon_Prototype_functions.cpp BP_EB_Weapon_functions.cpp
  BP_PlayerCharacter_functions.cpp CoreUObject_functions.cpp Engine_functions.cpp
  GameplayAbilities_functions.cpp island_arrival_cinematic_trigger_bp_functions.cpp)
list(TRANSFORM MP_SDK_SOURCES PREPEND "${MP_SDK_DIR}/SDK/")

set(MP_SOURCES
  dllmain.cpp
  ../tools/CatalogExporter/StoreRuntimeExporter.cpp
  MinHook/buffer.c MinHook/hde/hde32.c MinHook/hde/hde64.c MinHook/hook.c MinHook/trampoline.c
  Networking.cpp
  MysticParadox.rc)
# Optional player-hosted mode and its anti-cheat module, with the same conditions as the .vcxproj.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/p2p/PlayerHost.cpp")
  list(APPEND MP_SOURCES p2p/PlayerHost.cpp)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/anticheat/HuntGuard.cpp")
    list(APPEND MP_SOURCES anticheat/HuntGuard.cpp)
  else()
    list(APPEND MP_SOURCES p2p/AntiCheatHooksStub.cpp)
  endif()
else()
  list(APPEND MP_SOURCES P2PHooksStub.cpp)
endif()

add_library(MysticParadox SHARED ${MP_SOURCES} ${MP_SDK_SOURCES})
set_target_properties(MysticParadox PROPERTIES
  OUTPUT_NAME MysticParadox
  CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON           # .vcxproj: stdcpplatest
  MSVC_RUNTIME_LIBRARY MultiThreaded)               # .vcxproj: /MT (see the comment there)
target_include_directories(MysticParadox PRIVATE
  "${MP_GENERATED_INCLUDE}" "${MP_SDK_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(MysticParadox PRIVATE NDEBUG MYSTICPARADOX_EXPORTS _WINDOWS _USRDLL)
if(MSVC)  # MSVC and clang-cl: the .vcxproj Release|x64 switches
  target_compile_options(MysticParadox PRIVATE
    $<$<COMPILE_LANGUAGE:C,CXX>:/W3 /Gy /Oi /sdl /permissive- /EHsc /bigobj>)
  target_link_options(MysticParadox PRIVATE /SUBSYSTEM:WINDOWS /OPT:REF /OPT:ICF)
endif()
